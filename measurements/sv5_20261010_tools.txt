import sys, re
root = sys.argv[1]


def EOL(s):
    return chr(13) + chr(10) if (chr(13) + chr(10)) in s else chr(10)


def load(rel):
    raw = open(root + '/' + rel, 'rb').read()
    bom = raw.startswith(b'\xef\xbb\xbf')
    s = raw.decode('utf-8-sig')
    return s, bom


def save(rel, s, bom):
    open(root + '/' + rel, 'wb').write((b'\xef\xbb\xbf' if bom else b'') + s.encode('utf-8'))


def rep(rel, old, new):
    s, bom = load(rel)
    crlf = '\r\n' in s
    if crlf:
        old = old.replace('\n', '\r\n'); new = new.replace('\n', '\r\n')
    assert s.count(old) == 1, (rel, old[:80], s.count(old))
    save(rel, s.replace(old, new), bom)


def lines(rel, a, b, new):
    """行 a..b（1始まり・両端含む）を new（改行つき複数行）で置き換える。先頭行の先頭が old_head と一致することを呼び手が確かめる。"""
    s, bom = load(rel)
    crlf = '\r\n' in s
    ls = s.split('\r\n' if crlf else '\n')
    nl = new.split('\n')
    ls[a - 1:b] = nl
    save(rel, ('\r\n' if crlf else '\n').join(ls), bom)
